Overview

Adds t.amount to migrations.

Polymorphic amounts create *_atomic and *_currency columns. Fixed currency amounts create only the atomic column.

Examples:

create_table :holdings do |t|
  t.amount :balance
  t.amount :fee, currency: :SOL
  t.amount :reserve, precision: 40
end

def amount(name, precision: 78, currency: nil, **options)
  default = options.delete(:default)
  null = options.key?(:null) ? options[:null] : nil
  comment = options[:comment]

  defaults = normalize_defaults(default, currency)

  decimal_options = {
    precision: precision,
    scale: 0,
    default: defaults[:atomic],
    comment: comment
  }
  decimal_options[:null] = null unless null.nil?
  decimal(name_to_atomic(name), **decimal_options.compact)

  return if currency

  string_options = {
    limit: 10,
    default: defaults[:currency],
    comment: comment
  }
  string_options[:null] = null unless null.nil?
  string(name_to_currency(name), **string_options.compact)
end
