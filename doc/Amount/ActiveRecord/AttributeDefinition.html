def query_relation(model, value)
  amount = type.cast(value)
  relation = model.where(atomic_column => amount.atomic)
  fixed_currency? ? relation : relation.where(currency_column => amount.symbol.to_s)
end

def validate(record)
  add_assignment_error(record)

  atomic = record.read_attribute(atomic_column)
  return if atomic.nil? && fixed_currency?

  if fixed_currency?
    validate_deserialization(record, atomic, currency)
    return
  end

  currency_value = record.read_attribute(currency_column)
  if atomic.nil? != currency_value.nil?
    record.errors.add(name, "must set both atomic and currency or neither")
    return
  end

  return if atomic.nil? && currency_value.nil?

  validate_deserialization(record, atomic, currency_value)
end

def value_in_database(record)
  values = component_columns.to_h do |column|
    [column, record.attribute_in_database(column)]
  end

  build_from_values(values)
end

def write(record, value)
  if value.nil?
    record.write_attribute(atomic_column, nil)
    record.write_attribute(currency_column, nil) unless fixed_currency?
    return nil
  end

  record.write_attribute(atomic_column, value.atomic.to_s)
  record.write_attribute(currency_column, value.symbol.to_s) unless fixed_currency?
  value
end
